 (function(){
  // Render charts on juridico.php using Chart.js and available APIs
  const metricsApi = '/sistema_vendas/public/api/juridico_metrics.php';
  const processesApi = '/sistema_vendas/public/api/processes.php';

  function safeFetch(url){ return fetch(url,{credentials:'same-origin'}).then(r=>r.json()).catch(e=>{console.error(url,e);return null}); }

  function fmtDate(d){ if(!d) return ''; try{ return new Date(d).toLocaleDateString(); }catch(e){return d;} }

  // debug panel helpers
  let __jc_dbg = null;
  function createDebugPanel(){
    if (__jc_dbg) return __jc_dbg;
    const dbg = document.createElement('div'); dbg.id = 'juridicoChartsDebug';
    dbg.style.cssText = 'position:fixed; right:14px; top:14px; z-index:9999; max-width:360px; padding:10px 12px; color:#cfeeff; background:linear-gradient(180deg, rgba(7,18,30,0.9), rgba(6,14,26,0.85)); border:1px solid rgba(96,165,250,0.08); border-radius:8px; font-family:Inter,system-ui,Segoe UI,Arial; font-size:13px; box-shadow:0 6px 18px rgba(2,6,23,0.6)';
    dbg.innerHTML = '<strong style="display:block;margin-bottom:6px">Charts debug</strong>' +
      '<div id="jc_status_chart">Chart.js: â</div>' +

      '<div id="jc_status_processes">Processes API: â</div>' +
      '<div id="jc_status_render">Render: â</div>';
    document.body.appendChild(dbg);
    __jc_dbg = dbg;
    return dbg;
  }
  function setDbg(id, text){ try{ const dbg=createDebugPanel(); const el = dbg.querySelector('#jc_status_'+id); if (el) el.textContent = (id==='chart' ? 'Chart.js: ' : id==='metrics' ? 'Metrics API: ' : id==='processes' ? 'Processes API: ' : 'Render: ') + text; }catch(e){} }

  async function loadAndRender(){
    // ensure canvases have explicit pixel heights so Chart.js can compute sizes
    try{
      ['statusDonut','deadlinesWeekChart','hearingsLineChart','byLawyerBar','typesBar','completionGauge'].forEach(id=>{
        const c = document.getElementById(id);
        if (c && c.tagName === 'CANVAS') c.height = c.height || 260;
      });
    }catch(e){ console.warn('prepare canvases failed', e); }

    const m = await safeFetch(metricsApi);
    const all = await safeFetch(processesApi);
    const processes = (all && all.data) ? all.data : (m && m.data && m.data.by_lawyer ? [] : []);

    // update persistent debug panel with step statuses
    try{
      setDbg('chart', window.Chart ? 'loaded' : 'not loaded');
      setDbg('metrics', (m && m.success) ? ('ok â ' + Object.keys(m.data||{}).join(',')) : 'failed');
      const procStatus = Array.isArray(processes) ? (processes.length + ' items') : (all && all.error ? 'error' : 'not fetched');
      setDbg('processes', procStatus);
      setDbg('render', 'pending');
      console.log('juridico debug', {metrics:m, processes:all});
    }catch(e){ console.warn('debug panel failed', e); }

    try{
      // STATUS donut
      const statusOrder = ['ativo','em andamento','aguardando decisÃ£o','finalizado','urgente','arquivado'];
      const statusMap = { ativo:0, 'em andamento':0, 'aguardando decisÃ£o':0, finalizado:0, urgente:0, arquivado:0 };
      if (processes && processes.length){
        processes.forEach(p=>{
          const st = (p.status||'').toLowerCase();
          if (statusMap.hasOwnProperty(st)) statusMap[st]++;
          else statusMap['em andamento'] = (statusMap['em andamento']||0)+0;
          const urgent = (p.proximo_prazo && (new Date(p.proximo_prazo) - new Date() <= 3*24*3600*1000)) || (p.alerts && p.alerts.toString().toLowerCase().includes('urg'));
          if (urgent) statusMap['urgente']++;
        });
      } else if (m && m.data){
        statusMap['ativo'] = m.data.active_count || 0;
        statusMap['urgente'] = (m.data.urgent||[]).length || 0;
      }
      const statusLabels = ['Ativo','Em andamento','Aguardando decisÃ£o','Finalizado','Urgente','Arquivado'];
      const statusData = statusLabels.map((lab,i)=> statusMap[Object.keys(statusMap)[i]] || 0);
      const statusCtx = document.getElementById('statusDonut');
      if (statusCtx && window.Chart){
        new Chart(statusCtx.getContext('2d'), { type: 'doughnut', data: { labels: statusLabels, datasets:[{ data: statusData, backgroundColor: ['#60a5fa','#3b82f6','#2563eb','#1e40af','#93c5fd','#60a5fa'], borderColor: '#071427', borderWidth: 2 }] }, options: { elements: { arc: { borderColor: '#071427' }, point: { borderColor: '#071427' } }, plugins:{legend:{position:'bottom', labels:{color:'#cfeeff'}}, tooltip:{backgroundColor:'#0b2240'} }, responsive:true } });
      }

      // Prazos por semana (Mon-Fri)
      const weekdays = ['Segunda','TerÃ§a','Quarta','Quinta','Sexta'];
      const weekCounts = [0,0,0,0,0];
      if (processes && processes.length){
        processes.forEach(p=>{
          if (!p.proximo_prazo) return;
          const d = new Date(p.proximo_prazo);
          const day = d.getDay();
          if (day>=1 && day<=5) weekCounts[day-1]++;
        });
      } else if (m && m.data && Array.isArray(m.data.deadlines_week)){
        m.data.deadlines_week.forEach(x=>{ if (x.proximo_prazo){ const d=new Date(x.proximo_prazo); const day=d.getDay(); if (day>=1 && day<=5) weekCounts[day-1]++; } });
      }
      const deadlinesCtx = document.getElementById('deadlinesWeekChart');
      if (deadlinesCtx && window.Chart){
        new Chart(deadlinesCtx.getContext('2d'), { type: 'bar', data: { labels: weekdays, datasets:[{ label:'Prazos', data: weekCounts, backgroundColor:'#3b82f6', borderColor:'#071427', borderWidth:1 }] }, options:{ elements:{ point:{ borderColor:'#071427' } }, responsive:true, plugins:{legend:{display:false, labels:{color:'#cfeeff'}} } });
      }

      // AudiÃªncias por mÃªs (Jan-Dec)
      const months = ['Jan','Fev','Mar','Abr','Mai','Jun','Jul','Ago','Set','Out','Nov','Dez'];
      const monthsCounts = new Array(12).fill(0);
      if (processes && processes.length){
        processes.forEach(p=>{
          const text = ((p.observacoes||'')+ ' ' + (p.alerts||'')).toLowerCase();
          if (!text.includes('audi')) return;
          if (!p.proximo_prazo) return;
          const d = new Date(p.proximo_prazo); monthsCounts[d.getMonth()]++;
        });
      } else if (m && m.data && Array.isArray(m.data.hearings_month)){
        m.data.hearings_month.forEach(x=>{ if (x.proximo_prazo){ const d=new Date(x.proximo_prazo); monthsCounts[d.getMonth()]++; } });
      }
      const hearingsCtx = document.getElementById('hearingsLineChart');
      if (hearingsCtx && window.Chart){
        new Chart(hearingsCtx.getContext('2d'), { type:'line', data:{ labels: months, datasets:[{ label:'AudiÃªncias', data: monthsCounts, borderColor:'#60a5fa', backgroundColor:'rgba(96,165,250,0.12)', pointBackgroundColor:'#60a5fa', pointBorderColor:'#071427', fill:true }] }, options:{ elements:{ point:{ borderColor:'#071427' } }, responsive:true, plugins:{legend:{labels:{color:'#cfeeff'}}} } });
      }

      // Processos por advogado
      const byLawyerLabels = [];
      const byLawyerData = [];
      if (m && m.data && Array.isArray(m.data.by_lawyer) && m.data.by_lawyer.length){
        m.data.by_lawyer.forEach(r=>{ byLawyerLabels.push(r.nome || ('ID '+r.user_id)); byLawyerData.push(Number(r.total)||0); });
      } else if (processes && processes.length){
        const map = {};
        processes.forEach(p=>{ const name = p.responsavel_user_id || 'Sem responsÃ¡vel'; map[name] = (map[name]||0)+1; });
        Object.keys(map).sort((a,b)=>map[b]-map[a]).forEach(k=>{ byLawyerLabels.push(k); byLawyerData.push(map[k]); });
      }
      const byLawyerCtx = document.getElementById('byLawyerBar');
      if (byLawyerCtx && window.Chart){ new Chart(byLawyerCtx.getContext('2d'), { type:'bar', data:{ labels: byLawyerLabels.slice(0,10), datasets:[{ label:'Processos', data: byLawyerData.slice(0,10), backgroundColor:'#2563EB', borderColor:'#071427', borderWidth:1 }] }, options:{ elements:{ point:{ borderColor:'#071427' } }, indexAxis:'y', responsive:true, plugins:{legend:{display:false, labels:{color:'#cfeeff'}}} } }); }

      // Types of action frequency
      const typeMap = {}; const topTypes = ['Trabalhista','CÃ­vel','PrevidenciÃ¡rio','FamÃ­lia','Empresarial'];
      if (processes && processes.length){ processes.forEach(p=>{ const t = (p.tipo_acao||'').toString() || 'Outro'; typeMap[t] = (typeMap[t]||0)+1; }); }
      const typesLabels = Object.keys(typeMap).length ? Object.keys(typeMap) : topTypes;
      const typesData = typesLabels.map(l=> typeMap[l]||0 );
      const typesCtx = document.getElementById('typesBar'); if (typesCtx && window.Chart){ new Chart(typesCtx.getContext('2d'), { type:'bar', data:{ labels: typesLabels.slice(0,8), datasets:[{ data: typesData.slice(0,8), backgroundColor:'#60a5fa', borderColor:'#071427', borderWidth:1 }] }, options:{ elements:{ point:{ borderColor:'#071427' } }, responsive:true, plugins:{legend:{display:false, labels:{color:'#cfeeff'}}} } }); }

      // Completion rate (gauge-like using doughnut)
      let closedThisMonth = 0, newThisMonth = 0;
      const now = new Date(); const startMonth = new Date(now.getFullYear(), now.getMonth(), 1);
      if (processes && processes.length){
        processes.forEach(p=>{
          const created = p.created_at ? new Date(p.created_at) : null;
          const status = (p.status||'').toLowerCase();
          const closed = (status === 'encerrado' || status === 'arquivado' || status === 'finalizado');
          if (created && created >= startMonth) newThisMonth++;
          const updated = p.updated_at ? new Date(p.updated_at) : null;
          if (closed && updated && updated >= startMonth) closedThisMonth++;
        });
      }
      const gaugeCtx = document.getElementById('completionGauge');
      const totalForGauge = Math.max(newThisMonth, 1);
      const percent = Math.round((closedThisMonth / (totalForGauge || 1)) * 100);
      if (gaugeCtx && window.Chart){ new Chart(gaugeCtx.getContext('2d'), { type:'doughnut', data:{ labels:['Finalizados','Restantes'], datasets:[{ data:[closedThisMonth, Math.max(0,newThisMonth-closedThisMonth)], backgroundColor:['#2563EB','#93c5fd'], borderColor:'#071427', borderWidth:2 }] }, options:{ elements:{ arc:{ borderColor:'#071427' } }, cutout: '80%', responsive:true, plugins:{legend:{display:false, labels:{color:'#cfeeff'}}} } }); }
      const labelEl = document.getElementById('completionLabel'); if (labelEl) labelEl.textContent = `${closedThisMonth} finalizados / ${newThisMonth} novos este mÃªs (${percent}%)`;

      // Processes without update (cards + ranking)
      const staleBuckets = { '7':'>7 dias', '15':'>15 dias', '30':'>30 dias' };
      const staleCounts = { '7':0,'15':0,'30':0 };
      const nowTs = Date.now();
      if (processes && processes.length){
        processes.forEach(p=>{
          const ult = p.ultima_movimentacao ? new Date(p.ultima_movimentacao).getTime() : (p.created_at ? new Date(p.created_at).getTime() : 0);
          const days = Math.floor((nowTs - ult) / (24*3600*1000));
          if (days > 30) staleCounts['30']++;
          if (days > 15) staleCounts['15']++;
          if (days > 7) staleCounts['7']++;
        });
      } else if (m && m.data && Array.isArray(m.data.no_update)){
        m.data.no_update.forEach(p=>{ const ult = p.ultima_movimentacao ? new Date(p.ultima_movimentacao).getTime() : 0; const days = Math.floor((nowTs - ult)/(24*3600*1000)); if (days>30) staleCounts['30']++; if (days>15) staleCounts['15']++; if (days>7) staleCounts['7']++; });
      }
      const staleEl = document.getElementById('staleCards'); if (staleEl){ staleEl.innerHTML = ''; Object.keys(staleBuckets).forEach(k=>{ const div = document.createElement('div'); div.className='mb-2'; div.innerHTML = `<div style="font-weight:600">${staleBuckets[k]}</div><div style="font-size:20px;color:#fff">${staleCounts[k]}</div>`; staleEl.appendChild(div); }); }

      // Top lawyers small list
      const topEl = document.getElementById('topLawyers'); if (topEl){ topEl.innerHTML=''; if (byLawyerLabels.length){ byLawyerLabels.slice(0,10).forEach((lab,i)=>{ const w = document.createElement('div'); w.className='mb-1'; w.innerHTML = `<div style="display:flex;justify-content:space-between"><span>${lab}</span><span style="font-weight:600">${byLawyerData[i]||0}</span></div>`; topEl.appendChild(w); }); } }

      // mark render OK
      try{ setDbg('render', 'ok'); }catch(e){}
    }catch(err){
      console.error('charts render error', err);
      try{ setDbg('render', 'failed: ' + (err && err.message ? err.message : String(err))); }catch(e){}
      // show simple message in each area
      ['statusDonut','deadlinesWeekChart','hearingsLineChart','byLawyerBar','typesBar','completionGauge'].forEach(id=>{
        const el = document.getElementById(id);
        if (!el) return;
        if (el.tagName === 'CANVAS'){
          const wrap = el.parentElement; if (wrap) wrap.innerHTML = '<div style="padding:24px;color:#9fb6d9">Erro ao renderizar grÃ¡fico</div>';
        } else {
          el.innerHTML = '<div style="padding:24px;color:#9fb6d9">Erro ao renderizar grÃ¡fico</div>';
        }
      });
    }

  }

  // load Chart.js then render
  function loadChartJsAndRun(){
    if (window.Chart) return loadAndRender();
    const s = document.createElement('script'); s.src = 'https://cdn.jsdelivr.net/npm/chart.js@4.3.0/dist/chart.umd.min.js'; s.onload = loadAndRender; document.head.appendChild(s);
  }

  if (document.readyState==='loading') document.addEventListener('DOMContentLoaded', loadChartJsAndRun); else loadChartJsAndRun();

})();
